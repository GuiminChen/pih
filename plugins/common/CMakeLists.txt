# CPU-only offline tooling uses the same tokenizer as model plugins, with no
# CUDA, Python runtime, scheduler, or monolithic model archive dependency.
find_package(ICU REQUIRED COMPONENTS uc i18n)
add_executable(pih-tokenize tokenize_main.cpp bytelevel_tokenizer.cpp deepseek_semantic_artifact.cpp)
target_compile_features(pih-tokenize PRIVATE cxx_std_20)
target_link_libraries(pih-tokenize PRIVATE pih_core_primitives ICU::uc ICU::i18n)
pih_assert_direct_link_allowlist(pih-tokenize pih_core_primitives ICU::uc ICU::i18n)
if(MSVC)
    target_compile_options(pih-tokenize PRIVATE /utf-8)
endif()
install(TARGETS pih-tokenize RUNTIME DESTINATION bin)
add_executable(pih-qwen-format ../model-qwen3/format_main.cpp ../model-qwen3/tokenizer.cpp text_json.cpp bytelevel_tokenizer.cpp)

if(MSVC)
    target_compile_options(pih-qwen-format PRIVATE /utf-8)
endif()
install(TARGETS pih-qwen-format RUNTIME DESTINATION bin)
